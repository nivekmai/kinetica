cmake_minimum_required(VERSION 3.22)
project(kinetica_voice LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../../third_party/llama.cpp llama EXCLUDE_FROM_ALL)

add_library(kinetica_voice SHARED voice_cleanup.cpp)
target_link_libraries(kinetica_voice PRIVATE llama)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(kinetica_voice PRIVATE ${JNI_INCLUDE_DIRS})
endif()
